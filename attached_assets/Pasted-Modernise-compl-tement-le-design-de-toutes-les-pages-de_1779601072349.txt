## TOGGLE DARK/LIGHT MODE

### Crée src/contexts/ThemeContext.jsx :
import { createContext, useContext, useState, useEffect } 
  from 'react';

const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [isDark, setIsDark] = useState(
    localStorage.getItem('theme') !== 'light'
  );

  useEffect(() => {
    document.documentElement.classList.toggle('dark', isDark);
    localStorage.setItem('theme', isDark ? 'dark' : 'light');
  }, [isDark]);

  return (
    <ThemeContext.Provider value={{ isDark, setIsDark }}>
      {children}
    </ThemeContext.Provider>
  );
}

export const useTheme = () => useContext(ThemeContext);

### Composant Toggle réutilisable :
export function ThemeToggle() {
  const { isDark, setIsDark } = useTheme();
  
  return (
    <button
      onClick={() => setIsDark(!isDark)}
      className={`relative w-14 h-7 rounded-full 
        transition-all duration-300 
        ${isDark ? 'bg-orange-500' : 'bg-gray-200'}`}
    >
      <div className={`absolute top-0.5 w-6 h-6 
        bg-white rounded-full shadow-md
        transition-all duration-300 flex items-center 
        justify-center text-sm
        ${isDark ? 'left-7' : 'left-0.5'}`}>
        {isDark ? '🌙' : '☀️'}
      </div>
    </button>
  );
}
