## ÉTAPE 2 — Système de vérification automatique des résultats

Crée src/services/resultatService.js :

export const verifierResultat = async (candidature) => {
  const { 
    urlResultats, 
    numeroCandidat, 
    nomComplet,
    dateNaissance 
  } = candidature;

  try {
    // Appel vers ton backend pour scraper le site officiel
    const response = await fetch('/api/verifier-resultat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        url: urlResultats,
        numeroCandidat,
        nomComplet,
        dateNaissance
      })
    });

    const data = await response.json();
    return data;

  } catch (error) {
    console.error('Erreur vérification résultat:', error);
    return { statut: 'erreur', message: 'Impossible de vérifier' };
  }
};

// Vérifie automatiquement tous les jours si résultats disponibles
export const verifierToutesCandidatures = async (candidatures) => {
  const resultats = [];
  
  for (const c of candidatures) {
    // Vérifie seulement si date publication est passée
    const datePubli = new Date(c.datePublication);
    const aujourd_hui = new Date();
    
    if (datePubli <= aujourd_hui && c.statut === 'en_attente') {
      const resultat = await verifierResultat(c);
      resultats.push({ id: c.id, ...resultat });
      
      // Si résultat trouvé, envoie notification
      if (resultat.statut === 'admis') {
        await envoyerNotificationAdmission(c.nomConcours, true);
      } else if (resultat.statut === 'refuse') {
        await envoyerNotificationAdmission(c.nomConcours, false);
      }
    }
  }
  
  return resultats;
};

// Notification du résultat
const envoyerNotificationAdmission = async (nomConcours, admis) => {
  new Notification(
    admis ? '🎉 Félicitations !' : '😔 Résultat disponible',
    {
      body: admis 
        ? `Tu es ADMIS au ${nomConcours} ! Ouvre l'app pour les détails.`
        : `Le résultat du ${nomConcours} est disponible. Courage !`,
      icon: '/icon-192x192.png'
    }
  );
};


## ÉTAPE 3 — Backend API (server.js ou api/verifier-resultat.js)

Crée l'endpoint qui va chercher les résultats :

import fetch from 'node-fetch';
import * as cheerio from 'cheerio';

export default async function handler(req, res) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  const { url, numeroCandidat, nomComplet, dateNaissance } = req.body;

  try {
    // Récupère la page des résultats
    const response = await fetch(url, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (compatible; CompetencesCI/1.0)'
      }
    });
    
    const html = await response.text();
    const $ = cheerio.load(html);
    
    // Cherche le numéro de candidat dans la page
    const pageText = $('body').text().toLowerCase();
    const numeroCherche = numeroCandidat.toLowerCase();
    const nomCherche = nomComplet.toLowerCase();
    
    // Vérifie si le candidat est dans la liste
    const estAdmis = pageText.includes(numeroCherche) || 
                     pageText.includes(nomCherche);
    
    if (estAdmis) {
      return res.json({
        statut: 'admis',
        message: 'Félicitations ! Vous êtes admis.',
        details: extraireDetails($, numeroCandidat)
      });
    } else {
      // Vérifie si la page contient des résultats 
      // (page chargée mais candidat absent)
      const pageContientResultats = 
        pageText.includes('admis') || 
        pageText.includes('résultat') ||
        pageText.includes('liste');
        
      if (pageContientResultats) {
        return res.json({
          statut: 'refuse',
          message: 'Votre nom ne figure pas sur la liste.'
        });
      } else {
        return res.json({
          statut: 'en_attente',
          message: 'Les résultats ne sont pas encore publiés.'
        });
      }
    }
    
  } catch (error) {
    return res.status(500).json({
      statut: 'erreur',
      message: 'Impossible d\'accéder au site officiel.'
    });
  }
}

// Extrait les détails supplémentaires si disponibles
function extraireDetails($, numeroCandidat) {
  try {
    // Cherche la ligne contenant le numéro de candidat
    let details = '';
    $('tr, li, p').each((i, el) => {
      const text = $(el).text();
      if (text.includes(numeroCandidat)) {
        details = text.trim();
      }
    });
    return details;
  } catch {
    return '';
  }
}


## ÉTAPE 4 — Vérification automatique quotidienne

Dans App.js, lance la vérification chaque jour :

import { verifierToutesCandidatures } from './services/resultatService';

useEffect(() => {
  const verifierResultats = async () => {
    const candidatures = JSON.parse(
      localStorage.getItem('candidatures') || '[]'
    );
    
    if (candidatures.length > 0) {
      const resultats = await verifierToutesCandidatures(candidatures);
      
      // Met à jour les statuts
      resultats.forEach(r => {
        const index = candidatures.findIndex(c => c.id === r.id);
        if (index !== -1) {
          candidatures[index].statut = r.statut;
        }
      });
      
      localStorage.setItem('candidatures', JSON.stringify(candidatures));
    }
  };

  // Vérifie au démarrage
  verifierResultats();
  
  // Puis toutes les 24h
  const interval = setInterval(verifierResultats, 24 * 60 * 60 * 1000);
  return () => clearInterval(interval);
}, []);
