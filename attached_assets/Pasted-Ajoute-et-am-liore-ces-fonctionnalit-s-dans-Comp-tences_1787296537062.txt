### Crée src/hooks/useGlobalSearch.js :

import { useState, useCallback } from 'react';
import { useDebounce } from './useDebounce';

export function useGlobalSearch(
  ecoles, concours, articles
) {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState(null);
  const debouncedQuery = useDebounce(query, 300);

  useEffect(() => {
    if (!debouncedQuery || 
        debouncedQuery.length < 2) {
      setResults(null);
      return;
    }

    const q = debouncedQuery.toLowerCase();

    const ecolesFound = ecoles.filter(e =>
      e.nom?.toLowerCase().includes(q) ||
      e.ville?.toLowerCase().includes(q) ||
      e.domainesFormation?.some(f => 
        f.toLowerCase().includes(q)
      ) ||
      e.type?.toLowerCase().includes(q)
    );

    const concoursFound = concours.filter(c =>
      c.nom?.toLowerCase().includes(q) ||
      c.ministere?.toLowerCase().includes(q) ||
      c.niveau?.toLowerCase().includes(q) ||
      c.description?.toLowerCase().includes(q)
    );

    const articlesFound = articles.filter(a =>
      a.titre?.toLowerCase().includes(q) ||
      a.categorie?.toLowerCase().includes(q) ||
      a.resume?.toLowerCase().includes(q)
    );

    setResults({
      ecoles: ecolesFound.slice(0, 5),
      concours: concoursFound.slice(0, 5),
      articles: articlesFound.slice(0, 3),
      total: ecolesFound.length + 
             concoursFound.length + 
             articlesFound.length
    });
  }, [debouncedQuery, ecoles, concours, articles]);

  return { query, setQuery, results };
}

### Crée src/hooks/useDebounce.js :

import { useState, useEffect } from 'react';

export function useDebounce(value, delay) {
  const [debouncedValue, setDebouncedValue] = 
    useState(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

### Algorithme de recommandation :

const calculateRecommendations = (answers) => {
  const scores = {};
  
  ecoles.forEach(ecole => {
    let score = 0;
    
    // Compatibilité ville (20 points)
    if (ecole.ville === answers.ville) score += 20;
    
    // Compatibilité domaine (30 points)
    const domainesChoisis = answers.domaines || [];
    ecole.domainesFormation?.forEach(domaine => {
      if (domainesChoisis.some(d => 
        d.toLowerCase().includes(
          domaine.toLowerCase()
        )
      )) score += 30 / domainesChoisis.length;
    });
    
    // Compatibilité budget (25 points)
    const budgetMax = {
      'Faible': 200000,
      'Moyen': 600000,
      'Élevé': Infinity
    }[answers.budget] || Infinity;
    
    const frais = parseInt(
      ecole.fraisScolarite?.replace(/\D/g, '') || 0
    );
    if (frais <= budgetMax) score += 25;
    
    // Type établissement (15 points)
    if (answers.typeEtab === 'Les deux m\'intéressent' ||
        ecole.type === answers.typeEtab) score += 15;
    
    // Diplôme selon niveau (10 points)
    if (answers.niveau === 'BTS obtenu' &&
        ecole.diplomes?.some(d => 
          d.nom === 'Licence'
        )) score += 10;
    
    scores[ecole.id] = {
      ecole,
      score: Math.min(Math.round(score), 100),
      raisons: generateReasons(ecole, answers)
    };
  });
  
  return Object.values(scores)
    .sort((a, b) => b.score - a.score)
    .slice(0, 5);
};

const generateReasons = (ecole, answers) => {
  const raisons = [];
  
  if (ecole.ville === answers.ville) {
    raisons.push(`📍 Située à ${ecole.ville}, 
      ta ville`);
  }
  if (ecole.type === 'PUBLIC') {
    raisons.push('💰 Frais accessibles 
      (établissement public)');
  }
  answers.domaines?.forEach(domaine => {
    if (ecole.domainesFormation?.some(d => 
      d.toLowerCase().includes(
        domaine.toLowerCase()
      )
    )) {
      raisons.push(`✅ Propose ${domaine}`);
    }
  });
  
  return raisons;
};

### Hook useComparison :

Crée src/hooks/useComparison.js :

import { createContext, useContext, useState } 
  from 'react';

const ComparisonContext = createContext();

export function ComparisonProvider({ children }) {
  const [list, setList] = useState([]);

  const add = (ecole) => {
    if (list.length >= 3) {
      alert('Maximum 3 écoles à comparer');
      return;
    }
    if (!list.find(e => e.id === ecole.id)) {
      setList(prev => [...prev, ecole]);
    }
  };

  const remove = (id) => {
    setList(prev => prev.filter(e => e.id !== id));
  };

  const clear = () => setList([]);

  const isIn = (id) => list.some(e => e.id === id);

  return (
    <ComparisonContext.Provider value={{
      list, add, remove, clear, isIn
    }}>
      {children}
    </ComparisonContext.Provider>
  );
}

export const useComparison = () => 
  useContext(ComparisonContext);

### Crée src/hooks/useAuthGate.js :

export function useAuthGate() {
  const { user } = useAuth();
  const navigate = useNavigate();

  const requireAuth = (action) => {
    if (!user) {
      // Affiche une modal invitation 
      // à se connecter
      showAuthModal();
      return false;
    }
    action();
    return true;
  };

  return { requireAuth, isAuthenticated: !!user };
}
