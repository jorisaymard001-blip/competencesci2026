Ajoute un système de notifications push planifiées dans l'app 
"Compétences CI" avec Expo Notifications.

## INSTALLATION
npm install expo-notifications expo-device

## CONFIGURATION

Dans app.json, ajoute :
{
  "expo": {
    "plugins": [
      [
        "expo-notifications",
        {
          "icon": "./assets/notification-icon.png",
          "color": "#F97316"
        }
      ]
    ]
  }
}

// Récupère un message aléatoire selon l'heure
const getRandomMessage = (hour) => {
  const messages = MESSAGES[hour];
  return messages[Math.floor(Math.random() * messages.length)];
};

// Demande la permission
export const requestNotificationPermission = async () => {
  if (!Device.isDevice) return false;
  
  const { status: existingStatus } = 
    await Notifications.getPermissionsAsync();
  
  let finalStatus = existingStatus;
  
  if (existingStatus !== 'granted') {
    const { status } = await Notifications.requestPermissionsAsync();
    finalStatus = status;
  }
  
  return finalStatus === 'granted';
};

// Planifie toutes les notifications quotidiennes
export const scheduleAllNotifications = async () => {
  const hasPermission = await requestNotificationPermission();
  if (!hasPermission) return;

  // Annule les anciennes notifications
  await Notifications.cancelAllScheduledNotificationsAsync();

  const hours = [8, 12, 16, 20];

  for (const hour of hours) {
    const message = getRandomMessage(hour);
    
    await Notifications.scheduleNotificationAsync({
      content: {
        title: message.title,
        body: message.body,
        sound: true,
        priority: Notifications.AndroidNotificationPriority.HIGH,
        color: '#F97316',
      },
      trigger: {
        hour: hour,
        minute: 0,
        repeats: true, // Se répète chaque jour
      },
    });
  }

  console.log('✅ Notifications planifiées : 8h, 12h, 16h, 20h');
};

// Annule toutes les notifications (si user désactive)
export const cancelAllNotifications = async () => {
  await Notifications.cancelAllScheduledNotificationsAsync();
  console.log('🔕 Notifications annulées');
};

// Notification immédiate pour tester
export const sendTestNotification = async () => {
  await Notifications.scheduleNotificationAsync({
    content: {
      title: "🎓 Test Compétences CI",
      body: "Les notifications fonctionnent parfaitement !",
      color: '#F97316',
    },
    trigger: { seconds: 3 },
  });
};


## INTÉGRATION DANS App.js
Ajoute au démarrage de l'app :

import { scheduleAllNotifications } from './src/services/notificationService';
import * as Notifications from 'expo-notifications';
import { useEffect, useRef } from 'react';

// Dans ton composant App :
useEffect(() => {
  // Lance les notifications au démarrage
  scheduleAllNotifications();

  // Écoute les notifications reçues
  const subscription = Notifications.addNotificationReceivedListener(
    notification => {
      console.log('Notification reçue:', notification);
    }
  );

  // Écoute les clics sur notifications
  const responseSubscription = 
    Notifications.addNotificationResponseReceivedListener(response => {
      // Redirige vers la page Concours quand on clique
      navigation.navigate('Concours');
    });

  return () => {
    subscription.remove();
    responseSubscription.remove();
  };
}, []);


## PARAMÈTRE DANS LA PAGE PROFIL
Ajoute un toggle pour que l'utilisateur puisse activer/désactiver :

import { scheduleAllNotifications, cancelAllNotifications } 
  from '../services/notificationService';
import { Switch } from 'react-native';

// Dans ProfileScreen :
const [notificationsEnabled, setNotificationsEnabled] = useState(true);

const toggleNotifications = async (value) => {
  setNotificationsEnabled(value);
  if (value) {
    await scheduleAllNotifications();
  } else {
    await cancelAllNotifications();
  }
};

// UI à ajouter dans le profil :
<View style={styles.settingRow}>
  <Text style={styles.settingLabel}>🔔 Notifications</Text>
  <Switch
    value={notificationsEnabled}
    onValueChange={toggleNotifications}
    trackColor={{ false: '#374151', true: '#F97316' }}
    thumbColor={notificationsEnabled ? '#ffffff' : '#9CA3AF'}
  />
</View>
